\newcommand\clocks[3]{
    \newcommand\radius{1.7cm}
    \newcommand\hour{#1}
    \newcommand\minute{#2}
    \foreach \x [evaluate = \x as \y using int(\x/30)] in {30,60,...,360} {
        \draw [line width = \radius*0.1/10] (\x:\radius*9/10) -- (\x:\radius);
        \node (\x) at (90-\x:\radius*8/10) {\y};
    };
    \foreach \x in {0,6,...,360} {
        \draw [line width = \radius*0.1/20] (\x:\radius*9.5/10) -- (\x:\radius);
    };
    \draw [line width = \radius*0.1/10] (0:0cm) circle [radius = \radius];
    %\draw (0:0cm) circle [radius = \radius*0.4/10];
    \node (opoint) [draw,circle,minimum size=\radius*0.4/10,inner sep=1pt] at (0cm,0cm){};

    \ifthenelse{ \equal{#3}{1} }{
    \draw [line width = \radius*0.4/10] (90-\hour*30-30*\minute/60:-\radius*0.5/10) -- (90-\hour*30-30*\minute/60:\radius*5.5/10);
    \draw [line width = \radius*0.2/10] (90-\minute*6:-\radius*1/10) -- (90-\minute*6:\radius*7.5/10);
    \draw [line width = \radius*0.1/10] (90:-\radius*1.5/10) -- (90:\radius*8.5/10);}

    \ifthenelse{ \equal{#3}{1} }{
      \node [below=\radius*1.3 of opoint] {\underline{\qquad\qquad}:\underline{\qquad\qquad}};
    }{
      \node [below=\radius*1.2 of opoint] {\underline{#1}:\underline{#2}};
    }
}
